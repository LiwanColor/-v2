<script>

const raw =
'01|off_white_beige|أوف وايت بيج|#B8B6AA;' +
'02|warm_beige|بيج دافئ|#B2AD99;' +
'03|greige|بيج رمادي|#A09C89;' +
'04|taupe|تاوب بني رمادي|#8D826E;' +
'05|medium_gray_beige|رمادي بيج متوسط|#827C70;' +
'06|dark_brown_gray|بني رمادي داكن|#716A5A;' +
'07|dark_wood_brown|بني خشبي داكن|#58402D;' +
'08|brown_black|بني أسود|#251C17;' +
'09|mustard_gold|خردلي ذهبي|#6E4B22;' +
'10|burgundy|عنابي نبيذي|#2B1518;' +
'11|dark_olive|أخضر زيتي داكن|#333924;' +
'12|blue_gray|أزرق رمادي بترولي|#6A7B73;' +
'13|dark_blue_gray|أزرق رمادي داكن|#232936;' +
'14|deep_navy|كحلي شديد الغمقان|#111520;' +
'15|gray_off_white|أوف وايت رمادي|#ABABA3;' +
'16|warm_light_gray|رمادي فاتح دافئ|#99968F;' +
'17|medium_gray|رمادي متوسط|#605F5D;' +
'18|charcoal|فحمي داكن|#212226;' +
'19|black|أسود|#111112';


const colors = raw.split(';').map(x => {

  let [id,slug,name,hex] = x.split('|');

  return {
    id,
    slug,
    name,
    hex
  };

});


let outer = 4;
let inner = 10;


const $ = id => document.getElementById(id);


/* =========================
   SELECT OPTIONS
========================= */

function opt(el){

  el.innerHTML = colors.map(c => `
    <option value="${c.id}">
      ${c.id} — ${c.name}
    </option>
  `).join('');

}


/* =========================
   CURRENT PRODUCT
========================= */

function current(){

  const o = colors[outer - 1];
  const i = colors[inner - 1];

  const file =
    `outer_${o.id}_${o.slug}__inner_${i.id}_${i.slug}.jpg`;


  /* PRODUCT */

  $('product').src = 'images/' + file;

  $('product').alt =
    `سترة Liwan — خارجي ${o.name} وداخلي ${i.name}`;


  /* NAMES */

  $('outerName').textContent = o.name;
  $('innerName').textContent = i.name;


  /* OUTER */

  $('outerChosen').textContent =
    `${o.id} · ${o.name}`;

  $('outerHex').textContent =
    o.hex;

  $('outerCode').textContent =
    o.id;


  $('outerSwatch').style.background =
    o.hex;


  /* INNER */

  $('innerChosen').textContent =
    `${i.id} · ${i.name}`;

  $('innerHex').textContent =
    i.hex;

  $('innerCode').textContent =
    i.id;


  $('innerSwatch').style.background =
    i.hex;


  /* COMBINATION */

  $('combo').textContent =
    `${o.id} + ${i.id}`;


  /* DOWNLOAD */

  $('download').href =
    'images/' + file;

  $('download').download =
    `liwan-outer-${o.id}-inner-${i.id}.jpg`;


  /* ACTIVE PALETTE */

  document
    .querySelectorAll('.palette-item')
    .forEach(b => {

      b.classList.toggle(
        'active',
        +b.dataset.id === outer
      );

    });

}


/* =========================
   INITIALIZE
========================= */

opt($('outer'));
opt($('inner'));

$('outer').value = outer;
$('inner').value = inner;


/* =========================
   SELECT EVENTS
========================= */

$('outer').onchange = e => {

  outer = +e.target.value;

  current();

};


$('inner').onchange = e => {

  inner = +e.target.value;

  current();

};


/* =========================
   PALETTE
========================= */

$('palette').innerHTML =
  colors.map(c => `

    <button
      class="palette-item"
      data-id="${c.id}"
      title="تطبيق ${c.name} كخارجية"
    >

      <span
        class="swatch"
        style="background:${c.hex}"
      ></span>

      <span class="num">
        ${c.id}
      </span>

    </button>

  `).join('');


document
  .querySelectorAll('.palette-item')
  .forEach(b => {

    b.onclick = () => {

      outer = +b.dataset.id;

      $('outer').value = outer;

      current();

      window.scrollTo({
        top:0,
        behavior:'smooth'
      });

    };

  });


current();

</script>
